# owner:   qmxobs:qmxobs
# unit:    qmx-observability.service
# account: qmxobs (distinct non-qmx service account)

# /var/lib/qmx-observability is a named line item of registry:vps_disk_budget

# evidence value recorded at soak / provisioning — never a ratified constant
